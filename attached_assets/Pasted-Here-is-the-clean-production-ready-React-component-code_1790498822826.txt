Here is the clean, production-ready React component code to integrate the **Stale Offline-Queue Metric Card** into `ManagerAnalyticsDashboard.tsx`.

It connects directly to your existing `GET /api/v1/queues/counts` endpoint (or snapshot summary), applies the brand-consistent `portal-card` design system, and alerts managers when field officers have offline collections/applications sitting unsynced past the 7-day threshold.

---

### 1. The Stale Offline-Queue Metric Card Component (`StaleQueueMetricCard.tsx`)

```jsx
import React from 'react';
import { RefreshCw, AlertTriangle, ArrowRight, CheckCircle2 } from 'lucide-react';

interface StaleQueueMetricCardProps {
  staleCount: number;
  totalPendingQueueCount?: number;
  isLoading?: boolean;
  onNavigateToQueue: () => void;
}

export function StaleQueueMetricCard({
  staleCount = 0,
  totalPendingQueueCount = 0,
  isLoading = false,
  onNavigateToQueue,
}: StaleQueueMetricCardProps) {
  if (isLoading) {
    return (
      <div className="bg-slate-100 animate-pulse rounded-xl h-44 border border-slate-200 p-5" />
    );
  }

  const hasStaleItems = staleCount > 0;

  return (
    <div
      className={`rounded-xl border p-5 transition-all shadow-sm hover:shadow-md ${
        hasStaleItems
          ? 'bg-rose-50/40 border-rose-200 hover:border-rose-300'
          : 'bg-white border-slate-200 hover:border-slate-300'
      }`}
    >
      {/* Header */}
      <div className="flex items-center justify-between mb-3">
        <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider">
          Offline Queue Staleness
        </span>
        <div
          className={`p-2 rounded-lg ${
            hasStaleItems ? 'bg-rose-100 text-rose-700' : 'bg-emerald-100 text-emerald-700'
          }`}
        >
          {hasStaleItems ? (
            <AlertTriangle className="w-5 h-5" />
          ) : (
            <CheckCircle2 className="w-5 h-5" />
          )}
        </div>
      </div>

      {/* Primary Metric */}
      <div className="mb-4">
        <div className="flex items-baseline gap-2">
          <h3
            className={`text-2xl font-bold ${
              hasStaleItems ? 'text-rose-950' : 'text-slate-900'
            }`}
          >
            {staleCount} {staleCount === 1 ? 'Record' : 'Records'}
          </h3>
          {hasStaleItems && (
            <span className="text-xs px-2 py-0.5 rounded-full font-bold bg-rose-100 text-rose-700 border border-rose-200">
              &gt; 7 Days Unsynced
            </span>
          )}
        </div>
        
        <p className="text-xs text-slate-500 mt-1">
          {hasStaleItems ? (
            <span className="text-rose-800 font-medium">
              Requires field officer sync before audit cutoff.
            </span>
          ) : (
            <span>All pending queue items are under the 7-day age threshold.</span>
          )}
        </p>
      </div>

      {/* Action Footer */}
      <button
        type="button"
        onClick={onNavigateToQueue}
        className={`w-full flex items-center justify-between text-xs font-semibold pt-3 border-t transition-colors ${
          hasStaleItems
            ? 'text-rose-700 hover:text-rose-900 border-rose-200/60'
            : 'text-slate-600 hover:text-slate-900 border-slate-100'
        }`}
      >
        <span className="flex items-center gap-1.5">
          <RefreshCw className="w-3.5 h-3.5" />
          Inspect Offline Queue {totalPendingQueueCount > 0 ? `(${totalPendingQueueCount})` : ''}
        </span>
        <ArrowRight className="w-3.5 h-3.5" />
      </button>
    </div>
  );
}

```

### 2. Integration into `ManagerAnalyticsDashboard.tsx`

Add this 4th card to your executive metric grid alongside **Active Portfolio**, **Overdue / PAR**, and **Pending Approvals**.

```jsx
import React, { useEffect, useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { StaleQueueMetricCard } from './StaleQueueMetricCard';

export function ManagerAnalyticsDashboard({ selectedBranchId }: { selectedBranchId: string | null }) {
  const navigate = useNavigate();
  const [dashboardData, setDashboardData] = useState<any>(null);
  const [queueCounts, setQueueCounts] = useState<{ total: number; staleCount: number }>({ total: 0, staleCount: 0 });
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    async function loadDashboardMetrics() {
      setIsLoading(true);
      try {
        const branchQuery = selectedBranchId ? `?branchId=${selectedBranchId}` : '';
        
        // Fetch existing analytics snapshot + server-side queue counts in parallel
        const [reportsRes, queueCountsRes] = await Promise.all([
          fetch(`/api/v1/reports/manager${branchQuery}`, {
            headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
          }),
          fetch(`/api/v1/queues/counts${branchQuery}`, {
            headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
          })
        ]);

        const reportsJson = await reportsRes.json();
        const queueJson = await queueCountsRes.json();

        if (reportsJson.ok) setDashboardData(reportsJson.data);
        if (queueJson.ok || queueJson.success) {
          // Rule 3 compliance: Read counts strictly from server-side endpoint response
          setQueueCounts({
            total: queueJson.data?.pendingTotal || 0,
            staleCount: queueJson.data?.staleOfflineCount || reportsJson.data?.staleQueueRecords || 0,
          });
        }
      } catch (err) {
        console.error('Failed loading manager metrics:', err);
      } finally {
        setIsLoading(false);
      }
    }

    loadDashboardMetrics();
  }, [selectedBranchId]);

  return (
    <div className="space-y-6 p-6">
      <div className="flex items-center justify-between">
        <h1 className="text-xl font-bold text-slate-900">Manager Operations & Analytics</h1>
      </div>

      {/* 4-Column Executive Metrics Grid */}
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-5">
        
        {/* Card 1: Active Portfolio */}
        {/* ... Existing Portfolio Card ... */}

        {/* Card 2: PAR / Risk */}
        {/* ... Existing PAR Card ... */}

        {/* Card 3: Pending Approvals */}
        {/* ... Existing Approvals Card ... */}

        {/* Card 4: Stale Offline Queue (New Exposure Layer) */}
        <StaleQueueMetricCard
          staleCount={queueCounts.staleCount}
          totalPendingQueueCount={queueCounts.total}
          isLoading={isLoading}
          onNavigateToQueue={() => navigate('/collections/offline-queue')}
        />

      </div>
    </div>
  );
}

```

### Key System Rules Upheld

1. **No Client-Side Counting (Rule 3):** The `staleCount` and `totalPendingQueueCount` are calculated server-side inside `GET /api/v1/queues/counts` or `reporting.ts` and passed down as primitives.
2. **Design System Harmony:** Matches the shared `portal-card` border, typography, and hover interaction rules established in Phase 4.
3. **Zero Financial Logic Editing (Rule 1):** Operates purely as a read/exposure layer with zero edits to payment posting, schedule engine, or ledger.
